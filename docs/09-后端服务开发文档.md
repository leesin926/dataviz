```java
// 分布式锁工具
@Component
public class RedisLockHelper {

    @Resource
    private RedissonClient redissonClient;

    public <T> T executeWithLock(String lockKey, long waitTime, long leaseTime,
                                  TimeUnit unit, Supplier<T> supplier) {
        RLock lock = redissonClient.getLock(lockKey);
        try {
            boolean acquired = lock.tryLock(waitTime, leaseTime, unit);
            if (!acquired) {
                throw new BizException(500, "获取锁失败，请稍后重试");
            }
            return supplier.get();
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            throw new BizException(500, "获取锁被中断");
        } finally {
            if (lock.isHeldByCurrentThread()) {
                lock.unlock();
            }
        }
    }
}

// 缓存工具
@Component
public class CacheHelper {

    @Resource
    private RedisTemplate<String, Object> redisTemplate;

    public <T> T get(String key, Class<T> clazz) {
        Object value = redisTemplate.opsForValue().get(key);
        return clazz.isInstance(value) ? clazz.cast(value) : null;
    }

    public void set(String key, Object value, long timeout, TimeUnit unit) {
        redisTemplate.opsForValue().set(key, value, timeout, unit);
    }

    public boolean delete(String key) {
        return Boolean.TRUE.equals(redisTemplate.delete(key));
    }

    // 缓存穿透保护
    public <T> T getOrLoad(String key, long timeout, TimeUnit unit,
                            Class<T> clazz, Supplier<T> loader) {
        T cached = get(key, clazz);
        if (cached != null) return cached;
        synchronized (key.intern()) {
            cached = get(key, clazz);
            if (cached != null) return cached;
            cached = loader.get();
            if (cached != null) {
                set(key, cached, timeout, unit);
            }
            return cached;
        }
    }
}
```

```java
// JWT 工具
@Component
public class JwtHelper {

    @Value("${jwt.secret}")
    private String secret;

    @Value("${jwt.access-token-expire:7200}")
    private long accessTokenExpire;

    @Value("${jwt.refresh-token-expire:604800}")
    private long refreshTokenExpire;

    public String createAccessToken(LoginUser user) {
        return Jwts.builder()
                .setSubject(String.valueOf(user.getUserId()))
                .claim("tenantId", user.getTenantId())
                .claim("username", user.getUsername())
                .claim("permissions", user.getPermissions())
                .setIssuedAt(new Date())
                .setExpiration(new Date(System.currentTimeMillis() + accessTokenExpire * 1000))
                .signWith(Keys.hmacShaKeyFor(secret.getBytes()), SignatureAlgorithm.HS256)
                .compact();
    }

    public String createRefreshToken(Long userId) {
        return Jwts.builder()
                .setSubject(String.valueOf(userId))
                .claim("type", "refresh")
                .setIssuedAt(new Date())
                .setExpiration(new Date(System.currentTimeMillis() + refreshTokenExpire * 1000))
                .signWith(Keys.hmacShaKeyFor(secret.getBytes()), SignatureAlgorithm.HS256)
                .compact();
    }

    public Claims parseToken(String token) {
        return Jwts.parserBuilder()
                .setSigningKey(Keys.hmacShaKeyFor(secret.getBytes()))
                .build()
                .parseClaimsJws(token)
                .getBody();
    }

    public boolean validateToken(String token) {
        try {
            parseToken(token);
            return true;
        } catch (JwtException e) {
            return false;
        }
    }
}

// 认证拦截器
@Component
public class AuthInterceptor implements HandlerInterceptor {

    @Resource
    private JwtHelper jwtHelper;

    @Override
    public boolean preHandle(HttpServletRequest request, HttpServletResponse response,
                              Object handler) {
        String token = extractToken(request);
        if (StringUtils.isBlank(token)) {
            throw new BizException(ErrorCode.UNAUTHORIZED);
        }
        if (!jwtHelper.validateToken(token)) {
            throw new BizException(ErrorCode.TOKEN_INVALID);
        }
        Claims claims = jwtHelper.parseToken(token);
        LoginUser user = new LoginUser();
        user.setUserId(Long.valueOf(claims.getSubject()));
        user.setTenantId(claims.get("tenantId", Long.class));
        user.setUsername(claims.get("username", String.class));
        SecurityContextHolder.set(user);
        return true;
    }

    @Override
    public void afterCompletion(HttpServletRequest request, HttpServletResponse response,
                                 Object handler, Exception ex) {
        SecurityContextHolder.clear();
    }

    private String extractToken(HttpServletRequest request) {
        String header = request.getHeader("Authorization");
        if (StringUtils.isNotBlank(header) && header.startsWith("Bearer ")) {
            return header.substring(7);
        }
        return request.getParameter("access_token");
    }
}

// 权限注解
@Target(ElementType.METHOD)
@Retention(RetentionPolicy.RUNTIME)
public @interface RequiresPermission {
    String value();
}

// 权限拦截器
@Component
public class PermissionInterceptor implements HandlerInterceptor {

    @Resource
    private UserFeignClient userFeignClient;

    @Override
    public boolean preHandle(HttpServletRequest request, HttpServletResponse response,
                              Object handler) {
        if (!(handler instanceof HandlerMethod handlerMethod)) {
            return true;
        }
        RequiresPermission annotation = handlerMethod.getMethodAnnotation(RequiresPermission.class);
        if (annotation == null) return true;

        LoginUser user = SecurityContextHolder.get();
        AssertUtil.notNull(user, ErrorCode.UNAUTHORIZED);

        // 加载权限（从 Redis 缓存）
        if (user.getPermissions() == null) {
            R<Set<String>> result = userFeignClient.getUserPermissions(user.getUserId());
            user.setPermissions(result.getData());
        }
        AssertUtil.isTrue(user.getPermissions().contains(annotation.value()),
                ErrorCode.FORBIDDEN);
        return true;
    }
}
```

```java
// WebSocket 会话管理
@Component
public class WebSocketSessionManager {

    private final Map<String, Set<Session>> sessionMap = new ConcurrentHashMap<>();

    public void addSession(String roomId, Session session) {
        sessionMap.computeIfAbsent(roomId, k -> ConcurrentHashMap.newKeySet())
                .add(session);
    }

    public void removeSession(String roomId, Session session) {
        Set<Session> sessions = sessionMap.get(roomId);
        if (sessions != null) {
            sessions.remove(session);
            if (sessions.isEmpty()) {
                sessionMap.remove(roomId);
            }
        }
    }

    public void broadcast(String roomId, Object message) {
        Set<Session> sessions = sessionMap.get(roomId);
        if (sessions == null) return;
        String json = JsonUtils.toJson(message);
        sessions.forEach(session -> {
            try {
                session.getBasicRemote().sendText(json);
            } catch (IOException e) {
                log.error("WebSocket 发送失败", e);
            }
        });
    }

    public int getOnlineCount(String roomId) {
        Set<Session> sessions = sessionMap.get(roomId);
        return sessions != null ? sessions.size() : 0;
    }
}
```

```java
@Component
public class AuthGlobalFilter implements GlobalFilter, Ordered {

    private static final Set<String> WHITE_LIST = Set.of(
            "/auth/login", "/auth/register", "/auth/captcha",
            "/auth/refresh-token", "/open/callback"
    );

    @Resource
    private JwtHelper jwtHelper;

    @Override
    public Mono<Void> filter(ServerWebExchange exchange, GatewayFilterChain chain) {
        String path = exchange.getRequest().getURI().getPath();

        if (WHITE_LIST.stream().anyMatch(path::startsWith)) {
            return chain.filter(exchange);
        }

        String token = exchange.getRequest().getHeaders().getFirst("Authorization");
        if (StringUtils.isBlank(token) || !token.startsWith("Bearer ")) {
            exchange.getResponse().setStatusCode(HttpStatus.UNAUTHORIZED);
            return exchange.getResponse().setComplete();
        }

        token = token.substring(7);
        if (!jwtHelper.validateToken(token)) {
            exchange.getResponse().setStatusCode(HttpStatus.UNAUTHORIZED);
            return exchange.getResponse().setComplete();
        }

        Claims claims = jwtHelper.parseToken(token);
        ServerHttpRequest request = exchange.getRequest().mutate()
                .header("X-User-Id", claims.getSubject())
                .header("X-Tenant-Id", String.valueOf(claims.get("tenantId")))
                .header("X-Username", claims.get("username", String.class))
                .build();

        return chain.filter(exchange.mutate().request(request).build());
    }

    @Override
    public int getOrder() {
        return -100;
    }
}
```

```java
@RequiredArgsConstructor
public class AuthServiceImpl implements AuthService {

    private final UserMapper userMapper;
    private final UserRoleMapper userRoleMapper;
    private final RolePermissionMapper rolePermissionMapper;
    private final PermissionMapper permissionMapper;
    private final LoginLogMapper loginLogMapper;
    private final JwtHelper jwtHelper;
    private final CacheHelper cacheHelper;
    private final EventPublisher eventPublisher;
    private final PasswordEncoder passwordEncoder;

    @Override
    public LoginVO login(LoginDTO dto) {
        // 验证码校验
        verifyCaptcha(dto.getCaptchaKey(), dto.getCaptchaCode());

        // 查找用户
        SysUser user = userMapper.selectOne(
                new LambdaQueryWrapper<SysUser>()
                        .eq(SysUser::getUsername, dto.getUsername())
        );
        if (user == null) {
            recordLoginLog(dto.getUsername(), 1, false, "用户不存在");
            throw new BizException(ErrorCode.USER_NOT_FOUND);
        }
        if (user.getStatus() == 0) {
            recordLoginLog(dto.getUsername(), 1, false, "用户已禁用");
            throw new BizException(ErrorCode.USER_DISABLED);
        }
        if (!passwordEncoder.matches(dto.getPassword(), user.getPassword())) {
            recordLoginLog(dto.getUsername(), 1, false, "密码错误");
            throw new BizException(ErrorCode.PASSWORD_ERROR);
        }

        // 构建登录用户
        LoginUser loginUser = buildLoginUser(user);

        // 生成 Token
        String accessToken = jwtHelper.createAccessToken(loginUser);
        String refreshToken = jwtHelper.createRefreshToken(user.getId());

        // 缓存用户信息
        cacheHelper.set("login:user:" + user.getId(), loginUser, 2, TimeUnit.HOURS);

        // 记录登录日志
        recordLoginLog(dto.getUsername(), 1, true, "登录成功");

        // 更新登录信息
        userMapper.updateById(new SysUser() {{
            setId(user.getId());
            setLoginIp(dto.getIp());
            setLoginTime(LocalDateTime.now());
        }});

        // 发布事件
        eventPublisher.publish(new DomainEvent(
                EventTypes.USER_LOGIN, "User", String.valueOf(user.getId()),
                null, user.getTenantId(), user.getId(), LocalDateTime.now()
        ));

        LoginVO vo = new LoginVO();
        vo.setAccessToken(accessToken);
        vo.setRefreshToken(refreshToken);
        vo.setUserId(user.getId());
        vo.setUsername(user.getUsername());
        vo.setNickname(user.getNickname());
        vo.setAvatar(user.getAvatar());
        vo.setPermissions(loginUser.getPermissions());
        vo.setRoles(loginUser.getRoles());
        return vo;
    }

    @Override
    public void logout(String token) {
        if (StringUtils.isNotBlank(token) && token.startsWith("Bearer ")) {
            token = token.substring(7);
            Claims claims = jwtHelper.parseToken(token);
            Long userId = Long.valueOf(claims.getSubject());
            cacheHelper.delete("login:user:" + userId);
        }
    }

    @Override
    public TokenVO refreshToken(String refreshToken) {
        if (!jwtHelper.validateToken(refreshToken)) {
            throw new BizException(ErrorCode.TOKEN_INVALID);
        }
        Claims claims = jwtHelper.parseToken(refreshToken);
        if (!"refresh".equals(claims.get("type"))) {
            throw new BizException(ErrorCode.TOKEN_INVALID);
        }
        Long userId = Long.valueOf(claims.getSubject());
        SysUser user = userMapper.selectById(userId);
        AssertUtil.notNull(user, ErrorCode.USER_NOT_FOUND);

        LoginUser loginUser = buildLoginUser(user);
        String newAccessToken = jwtHelper.createAccessToken(loginUser);
        String newRefreshToken = jwtHelper.createRefreshToken(userId);

        TokenVO vo = new TokenVO();
        vo.setAccessToken(newAccessToken);
        vo.setRefreshToken(newRefreshToken);
        return vo;
    }

    private LoginUser buildLoginUser(SysUser user) {
        LoginUser loginUser = new LoginUser();
        loginUser.setUserId(user.getId());
        loginUser.setTenantId(user.getTenantId());
        loginUser.setUsername(user.getUsername());

        // 查询角色
        List<SysRole> roles = userRoleMapper.selectRolesByUserId(user.getId());
        loginUser.setRoles(roles.stream().map(SysRole::getRoleKey).collect(Collectors.toSet()));

        // 查询权限
        List<SysPermission> permissions = rolePermissionMapper.selectPermissionsByUserId(
                user.getId());
        loginUser.setPermissions(permissions.stream()
                .map(SysPermission::getPermissionKey)
                .collect(Collectors.toSet()));

        return loginUser;
    }

    private void recordLoginLog(String username, int loginType, boolean success, String message) {
        SysLoginLog log = new SysLoginLog();
        log.setUsername(username);
        log.setLoginType(loginType);
        log.setStatus(success ? 1 : 0);
        log.setMessage(message);
        loginLogMapper.insert(log);
    }
}
```

```java
@Component
public class DataSourceConnectionFactory {

    private final Map<Long, HikariDataSource> connectionPoolCache = new ConcurrentHashMap<>();

    public Connection getConnection(Datasource ds) {
        HikariDataSource pool = connectionPoolCache.computeIfAbsent(ds.getId(), id -> {
            HikariConfig config = new HikariConfig();
            config.setJdbcUrl(buildJdbcUrl(ds));
            config.setUsername(ds.getUsername());
            config.setPassword(DecryptUtil.decrypt(ds.getPassword()));
            config.setMaximumPoolSize(5);
            config.setMinimumIdle(1);
            config.setConnectionTimeout(10000);
            config.setIdleTimeout(300000);
            config.setMaxLifetime(600000);
            return new HikariDataSource(config);
        });
        try {
            return pool.getConnection();
        } catch (SQLException e) {
            throw new BizException(ErrorCode.DS_CONNECT_FAILED);
        }
    }

    public void testConnection(Datasource ds) {
        try (Connection conn = getConnection(ds)) {
            conn.isValid(5);
        } catch (SQLException e) {
            throw new BizException(ErrorCode.DS_CONNECT_FAILED);
        }
    }

    public void closePool(Long datasourceId) {
        HikariDataSource pool = connectionPoolCache.remove(datasourceId);
        if (pool != null && !pool.isClosed()) {
            pool.close();
        }
    }

    private String buildJdbcUrl(Datasource ds) {
        return switch (ds.getType().toUpperCase()) {
            case "MYSQL" -> String.format("jdbc:mysql://%s:%d/%s?useSSL=false&serverTimezone=Asia/Shanghai",
                    ds.getHost(), ds.getPort(), ds.getDatabaseName());
            case "POSTGRESQL" -> String.format("jdbc:postgresql://%s:%d/%s",
                    ds.getHost(), ds.getPort(), ds.getDatabaseName());
            case "CLICKHOUSE" -> String.format("jdbc:clickhouse://%s:%d/%s",
                    ds.getHost(), ds.getPort(), ds.getDatabaseName());
            case "ORACLE" -> String.format("jdbc:oracle:thin:@%s:%d:%s",
                    ds.getHost(), ds.getPort(), ds.getDatabaseName());
            case "SQLSERVER" -> String.format("jdbc:sqlserver://%s:%d;databaseName=%s",
                    ds.getHost(), ds.getPort(), ds.getDatabaseName());

@Service
@RequiredArgsConstructor
public class MetadataSyncService {

    private final DataSourceConnectionFactory connectionFactory;
    private final DatasourceMapper datasourceMapper;
    private final DatasourceMetadataMapper metadataMapper;

    public void syncMetadata(Long datasourceId) {
        Datasource ds = datasourceMapper.selectById(datasourceId);
        AssertUtil.notNull(ds, ErrorCode.DS_NOT_FOUND);

        try (Connection conn = connectionFactory.getConnection(ds)) {
            DatabaseMetaData metaData = conn.getMetaData();
            ResultSet tables = metaData.getTables(ds.getDatabaseName(), null, "%",
                    new String[]{"TABLE", "VIEW"});

            while (tables.next()) {
                String tableName = tables.getString("TABLE_NAME");
                String tableComment = tables.getString("REMARKS");

                // 获取列信息
                List<ColumnInfo> columns = new ArrayList<>();
                ResultSet columnsRs = metaData.getColumns(ds.getDatabaseName(), null,
                        tableName, "%");
                while (columnsRs.next()) {
                    ColumnInfo col = new ColumnInfo();
                    col.setName(columnsRs.getString("COLUMN_NAME"));
                    col.setType(columnsRs.getString("TYPE_NAME"));
                    col.setSize(columnsRs.getInt("COLUMN_SIZE"));
                    col.setNullable(columnsRs.getInt("NULLABLE") == 1);
                    col.setComment(columnsRs.getString("REMARKS"));
                    col.setPrimaryKey(false);
                    columns.add(col);
                }

                // 标记主键
                ResultSet pkRs = metaData.getPrimaryKeys(ds.getDatabaseName(), null, tableName);
                Set<String> pkColumns = new HashSet<>();
                while (pkRs.next()) {
                    pkColumns.add(pkRs.getString("COLUMN_NAME"));
                }
                columns.forEach(c -> c.setPrimaryKey(pkColumns.contains(c.getName())));

                // 保存元数据
                DatasourceMetadata metadata = new DatasourceMetadata();
                metadata.setDatasourceId(datasourceId);
                metadata.setTableName(tableName);
                metadata.setTableComment(tableComment);
                metadata.setColumnsJson(JsonUtils.toJson(columns));
                metadata.setSyncTime(LocalDateTime.now());
                metadataMapper.insert(metadata);
            }
        } catch (SQLException e) {
            throw new BizException(ErrorCode.DS_CONNECT_FAILED);
```

```java
@Service
@RequiredArgsConstructor
public class DagExecutor {

    private final OperatorFactory operatorFactory;
    private final ThreadPoolExecutor etlThreadPool;

    public DagResult execute(DagDefinition dag, Map<String, Object> params) {
        // 拓扑排序
        List<DagNode> sortedNodes = topologicalSort(dag);

        // 节点状态
        Map<String, NodeResult> nodeResults = new ConcurrentHashMap<>();
        Map<String, Integer> inDegree = calculateInDegree(dag);

        // 找到入度为 0 的节点
        Queue<DagNode> readyQueue = new LinkedList<>();
        for (DagNode node : sortedNodes) {
            if (inDegree.getOrDefault(node.getId(), 0) == 0) {
                readyQueue.add(node);
            }
        }

        DagResult result = new DagResult();
        result.setStartTime(LocalDateTime.now());

        while (!readyQueue.isEmpty()) {
            DagNode node = readyQueue.poll();

            // 并行执行当前就绪的节点
            CompletableFuture<NodeResult> future = CompletableFuture.supplyAsync(() -> {
                Operator operator = operatorFactory.getOperator(node.getType());
                NodeContext ctx = new NodeContext(node, params, nodeResults);
                return operator.execute(ctx);
            }, etlThreadPool);

            try {
                NodeResult nodeResult = future.get(30, TimeUnit.MINUTES);
                nodeResults.put(node.getId(), nodeResult);

                if (nodeResult.isSuccess()) {
                    // 减少后继节点入度
                    for (String nextId : dag.getEdges().stream()
                            .filter(e -> e.getSource().equals(node.getId()))
                            .map(DagEdge::getTarget)
                            .toList()) {
                        int newDegree = inDegree.get(nextId) - 1;
                        inDegree.put(nextId, newDegree);
                        if (newDegree == 0) {
                            readyQueue.add(dag.getNodeById(nextId));
                        }
                    }
                } else {
                    result.setStatus("FAILED");
                    result.setErrorMsg("节点 " + node.getName() + " 执行失败: "
                            + nodeResult.getErrorMsg());
                    return result;
                }
            } catch (TimeoutException e) {
                result.setStatus("FAILED");
                result.setErrorMsg("节点 " + node.getName() + " 执行超时");
                return result;
            } catch (Exception e) {
                result.setStatus("FAILED");
                result.setErrorMsg("节点 " + node.getName() + " 执行异常: " + e.getMessage());
                return result;
            }
        }

        result.setStatus("SUCCESS");
        result.setEndTime(LocalDateTime.now());
        result.setNodeResults(nodeResults);
        return result;
    }

    private List<DagNode> topologicalSort(DagDefinition dag) {
        List<DagNode> result = new ArrayList<>();
        Map<String, Integer> inDegree = calculateInDegree(dag);
        Queue<DagNode> queue = new LinkedList<>();

        for (DagNode node : dag.getNodes()) {
            if (inDegree.getOrDefault(node.getId(), 0) == 0) {
                queue.add(node);
            }
        }

        while (!queue.isEmpty()) {
            DagNode node = queue.poll();
            result.add(node);
            for (String nextId : dag.getEdges().stream()
                    .filter(e -> e.getSource().equals(node.getId()))
                    .map(DagEdge::getTarget).toList()) {
                int newDegree = inDegree.get(nextId) - 1;
                inDegree.put(nextId, newDegree);
                if (newDegree == 0) {
                    queue.add(dag.getNodeById(nextId));
                }
            }
        }

        if (result.size() != dag.getNodes().size()) {
            throw new BizException(400, "DAG 存在环路");
        }
        return result;
    }
}
```

### 11.1 查询引擎

```java
@Service
@RequiredArgsConstructor
public class QueryEngine {

    private final DatasetService datasetService;
    private final MetricService metricService;
    private final DimensionService dimensionService;
    private final DataSourceConnectionFactory connectionFactory;
    private final PermissionFilter permissionFilter;
    private final CacheHelper cacheHelper;

    public QueryResult execute(AnalysisQuery query) {
        // 1. 缓存检查
        String cacheKey = buildCacheKey(query);
        QueryResult cached = cacheHelper.get(cacheKey, QueryResult.class);
        if (cached != null) return cached;

        // 2. 解析查询
        Dataset dataset = datasetService.getById(query.getDatasetId());
        AssertUtil.notNull(dataset, ErrorCode.NOT_FOUND);

        // 3. 生成 SQL
        String sql = buildSql(query, dataset);

        // 4. 权限过滤
        sql = permissionFilter.filter(sql, query.getDatasetId());

        // 5. 执行查询
        QueryResult result = executeSql(dataset.getDatasourceId(), sql, query.getLimit());

        // 6. 缓存结果
        cacheHelper.set(cacheKey, result, 5, TimeUnit.MINUTES);

        return result;
    }

    private String buildSql(AnalysisQuery query, Dataset dataset) {
        StringBuilder sql = new StringBuilder();

        // SELECT
        sql.append("SELECT ");
        List<String> selectParts = new ArrayList<>();

        // 维度列
        for (DimensionRef dim : query.getDimensions()) {
            ModelDimension dimension = dimensionService.getById(dim.getDimensionId());
            selectParts.add(dimension.getColumnName() + " AS `" + dimension.getName() + "`");
        }

        // 指标列
        for (MetricRef metric : query.getMetrics()) {
            ModelMetric m = metricService.getById(metric.getMetricId());
            String aggExpr = buildAggExpression(m, metric);
            selectParts.add(aggExpr + " AS `" + m.getName() + "`");
        }

        sql.append(String.join(", ", selectParts));

        // FROM
        String fromClause = dataset.getType() == 2
                ? "(" + dataset.getSqlContent() + ") t"
                : dataset.getTableName() + " t";
        sql.append(" FROM ").append(fromClause);

        // JOIN (多表关联)
        if (dataset.getType() == 3 && dataset.getJoinConfig() != null) {
            sql.append(buildJoinClause(dataset.getJoinConfig()));
        }

        // WHERE
        if (query.getFilters() != null && !query.getFilters().isEmpty()) {
            sql.append(" WHERE ").append(buildWhereClause(query.getFilters()));
        }

        // GROUP BY
        if (!query.getDimensions().isEmpty()) {
            sql.append(" GROUP BY ");
            sql.append(query.getDimensions().stream()
                    .map(d -> {
                        ModelDimension dim = dimensionService.getById(d.getDimensionId());
                        return dim.getColumnName();
                    })
                    .collect(Collectors.joining(", ")));
        }

        // ORDER BY
        if (query.getOrders() != null && !query.getOrders().isEmpty()) {
            sql.append(" ORDER BY ");
            sql.append(query.getOrders().stream()
                    .map(o -> {
                        ModelMetric m = metricService.getById(o.getMetricId());
                        return "`" + m.getName() + "` " + o.getDirection();
                    })
                    .collect(Collectors.joining(", ")));
        }

        // LIMIT
        int limit = query.getLimit() != null ? query.getLimit() : 10000;
        sql.append(" LIMIT ").append(limit);

        return sql.toString();
    }

    private String buildAggExpression(ModelMetric metric, MetricRef ref) {
        if (StringUtils.isNotBlank(metric.getExpression())) {
            return metric.getExpression();
        }
        String func = metric.getAggFunction();
        String col = metric.getColumnName();
        return func + "(" + col + ")";
    }

    private QueryResult executeSql(Long datasourceId, String sql, Integer limit) {
        try (Connection conn = connectionFactory.getConnection(
                datasourceMapper.selectById(datasourceId));
             Statement stmt = conn.createStatement();
             ResultSet rs = stmt.executeQuery(sql)) {

            ResultSetMetaData metaData = rs.getMetaData();
            int columnCount = metaData.getColumnCount();

            List<ColumnMeta> columns = new ArrayList<>();
            for (int i = 1; i <= columnCount; i++) {
                ColumnMeta col = new ColumnMeta();
                col.setName(metaData.getColumnLabel(i));
                col.setType(metaData.getColumnTypeName(i));
                columns.add(col);
            }

            List<Map<String, Object>> rows = new ArrayList<>();
            int maxRows = limit != null ? limit : 10000;
            while (rs.next() && rows.size() < maxRows) {
                Map<String, Object> row = new LinkedHashMap<>();
                for (int i = 1; i <= columnCount; i++) {
                    row.put(columns.get(i - 1).getName(), rs.getObject(i));
                }
                rows.add(row);
            }

            QueryResult result = new QueryResult();
            result.setColumns(columns);
            result.setRows(rows);
            result.setTotal(rows.size());
            return result;
        } catch (SQLException e) {
            throw new BizException(500, "查询执行失败: " + e.getMessage());
        }
    }
}
```

```java
public class AlertCheckEngine {

    private final AlertRuleMapper ruleMapper;
    private final AlertEventMapper eventMapper;
    private final QueryEngine queryEngine;
    private final NotifyDispatcher notifyDispatcher;
    private final EventPublisher eventPublisher;

    @Scheduled(fixedDelay = 10000)
    public void checkAlerts() {
        List<AlertRule> rules = ruleMapper.selectDueRules(LocalDateTime.now());
        for (AlertRule rule : rules) {
            try {
                checkRule(rule);
            } catch (Exception e) {
                log.error("告警检查失败: ruleId={}", rule.getId(), e);
            }
        }
    }

    private void checkRule(AlertRule rule) {
        AlertCondition condition = JsonUtils.fromJson(rule.getConditionJson(),
                AlertCondition.class);

        // 执行检测查询
        QueryResult result = queryEngine.execute(condition.getQuery());
        Object currentValue = extractValue(result, condition.getValuePath());

        // 判断是否触发
        boolean triggered = evaluateCondition(currentValue, condition);

        if (triggered) {
            AlertEvent event = new AlertEvent();
            event.setRuleId(rule.getId());
            event.setTenantId(rule.getTenantId());
            event.setSeverity(rule.getSeverity());
            event.setTitle(rule.getName());
            event.setContent(buildAlertContent(rule, currentValue, condition));
            event.setTriggerValue(String.valueOf(currentValue));
            event.setThresholdValue(String.valueOf(condition.getThreshold()));
            event.setTriggerTime(LocalDateTime.now());
            eventMapper.insert(event);

            // 发送通知
            notifyDispatcher.dispatch(rule, event);

            // 更新规则
            ruleMapper.updateLastCheckTime(rule.getId(), LocalDateTime.now());
        }
    }

    private boolean evaluateCondition(Object value, AlertCondition condition) {
        double currentVal = Double.parseDouble(String.valueOf(value));
        double threshold = condition.getThreshold();
        return switch (condition.getOperator()) {
            case "GT" -> currentVal > threshold;
            case "GTE" -> currentVal >= threshold;
            case "LT" -> currentVal < threshold;
            case "LTE" -> currentVal <= threshold;
            case "EQ" -> Math.abs(currentVal - threshold) < 0.001;
            case "NEQ" -> Math.abs(currentVal - threshold) >= 0.001;
            default -> false;
        };
    }
}

                NotifyChannel.class);

                switch (channel.getType()) {
                    case "EMAIL" -> emailNotifier.send(channel, event);
                    case "WEBHOOK" -> webhookNotifier.send(channel, event);

                    case "DINGTALK" -> dingTalkNotifier.send(channel, event);
                    case "WECHAT" -> weChatNotifier.send(channel, event);
                }

                log.error("告警通知发送失败: channel={}, event={}", channel.getType(),
                        event.getId(), e);
```

```java
                %s
                
                参考示例：
                %s
                
                用户问题：%s
                
                请只返回 SQL 语句，不要其他解释。
                """, schemaDesc, examplesPrompt, naturalLanguage);

        // 调用 LLM
        String sql = llmClient.chat(prompt);
```

## 二十五、编码规范

| 规范项 | 说明 |
|--------|------|
| 包结构 | `com.dataviz.{service}.{controller/service/mapper/entity/dto/vo/config}` |
| 命名 | 类名大驼峰，方法/变量小驼峰，常量全大写下划线，数据库表名下划线 |
| Controller | 只做参数校验和调用 Service，返回 `R<T>` 统一格式 |
| Service | 业务逻辑层，事务控制在此层 |
| Mapper | 简单查询用 MyBatis-Plus 方法，复杂 SQL 用 XML |
| DTO/VO | DTO 用于入参，VO 用于出参，Entity 不直接暴露到 Controller |
| 异常处理 | 业务异常抛 `BizException`，由全局处理器统一捕获 |
| 日志 | 使用 `@Slf4j`，关键操作记录审计日志 |
| 事务 | `@Transactional` 标注在 Service 方法上，注意回滚异常 |
| 分页 | 统一使用 `PageQuery` / `PageResult` |
| 接口版本 | URL 前缀 `/api/v1/`，重大变更升级到 `/api/v2/` |
| 安全 | 密码 BCrypt 加密，敏感配置加密存储，SQL 防注入 |
