## 一、概述

### 1.1 定位

PC 桌面端基于 Electron 构建，面向需要离线编辑、本地文件处理、多窗口操作、硬件调用（投屏/打印机）等场景的用户。复用 PC 浏览器端的核心业务代码，通过 Electron 壳提供桌面特有能力。

### 1.2 核心能力

- 离线编辑（无网络环境）
- 本地文件拖拽导入
- 多窗口/多屏管理
- 系统托盘与全局快捷键
- 自动更新
- 本地缓存
- 硬件调用（投屏/打印机/摄像头）

### 4.1 主进程

#### 4.1.1 应用入口

```typescript
// electron/main.ts
import { app, BrowserWindow, Menu, Tray } from 'electron'
import { createWindow } from './window'
import { setupIpc } from './ipc'
import { initUpdater } from './services/updater'
import { initTray } from './services/tray'
import { initOffline } from './services/offline'

let mainWindow: BrowserWindow | null = null
let tray: Tray | null = null

app.whenReady().then(async () => {
  // 创建主窗口
  mainWindow = await createWindow()
  
  // 设置 IPC
  setupIpc(mainWindow)
  
  // 初始化托盘
  tray = initTray(mainWindow)
  
  // 初始化离线服务
  await initOffline()
  
  // 检查更新
  if (app.isPackaged) {
    initUpdater(mainWindow)
  }
})

app.on('window-all-closed', () => {
  if (process.platform !== 'darwin') {
    app.quit()
  }
})

app.on('activate', () => {
  if (BrowserWindow.getAllWindows().length === 0) {
    mainWindow = createWindow()
  }
})
```

#### 4.1.2 窗口管理

```typescript
// electron/window.ts
import { BrowserWindow, shell } from 'electron'
import path from 'path'

export async function createWindow(): Promise<BrowserWindow> {
  const win = new BrowserWindow({
    width: 1400,
    height: 900,
    minWidth: 1024,
    minHeight: 768,
    titleBarStyle: 'hiddenInset',
    frame: process.platform === 'darwin',
    webPreferences: {
      preload: path.join(__dirname, 'preload.js'),
      contextIsolation: true,
      nodeIntegration: false,
      webSecurity: true
    }
  })

  // 开发环境加载 devServer，生产环境加载本地文件
  if (process.env.NODE_ENV === 'development') {
    await win.loadURL('http://localhost:5173')
    win.webContents.openDevTools()
  } else {
    await win.loadFile(path.join(__dirname, '../dist/index.html'))
  }

  // 外部链接用浏览器打开
  win.webContents.setWindowOpenHandler(({ url }) => {
    shell.openExternal(url)
    return { action: 'deny' }
  })

  return win
}

// 创建多窗口
export function createAdditionalWindow(url: string): BrowserWindow {
  const win = new BrowserWindow({
    width: 1200,
    height: 800,
    parent: mainWindow,
    webPreferences: {
      preload: path.join(__dirname, 'preload.js'),
      contextIsolation: true
    }
  })
  
  win.loadURL(url)
  return win
}
```

### 4.3 离线功能

```typescript
// electron/services/offline.ts
import { app } from 'electron'
import path from 'path'
import fs from 'fs/promises'
import { ipcMain } from 'electron'

const CACHE_DIR = path.join(app.getPath('userData'), 'cache')

interface OfflineCache {
  dashboards: Map<string, any>
  screens: Map<string, any>
  datasources: Map<string, any>
  lastSync: number
}

let cache: OfflineCache = {
  dashboards: new Map(),
  screens: new Map(),
  datasources: new Map(),
  lastSync: 0
}

export async function initOffline() {
  // 加载缓存
  await loadCache()
  
  // 监听网络状态
  ipcMain.handle('offline:isOffline', () => {
    return !require('electron').net.isOnline()
  })
  
  ipcMain.handle('offline:getCacheStatus', () => {
    return {
      dashboards: cache.dashboards.size,
      screens: cache.screens.size,
      lastSync: cache.lastSync
    }
  })
}

async function loadCache() {
  try {
    const data = await fs.readFile(path.join(CACHE_DIR, 'offline.json'), 'utf-8')
    const parsed = JSON.parse(data)
    cache.dashboards = new Map(Object.entries(parsed.dashboards || {}))
    cache.screens = new Map(Object.entries(parsed.screens || {}))
    cache.lastSync = parsed.lastSync || 0
  } catch {
    // 缓存不存在，使用默认值
  }
}

async function saveCache() {
  await fs.mkdir(CACHE_DIR, { recursive: true })
  const data = {
    dashboards: Object.fromEntries(cache.dashboards),
    screens: Object.fromEntries(cache.screens),
    lastSync: cache.lastSync
  }
  await fs.writeFile(
    path.join(CACHE_DIR, 'offline.json'),
    JSON.stringify(data)
  )
}

// 缓存仪表板
export async function cacheDashboard(id: string, dashboard: any) {
  cache.dashboards.set(id, dashboard)
  await saveCache()
}

// 获取缓存的仪表板
export function getCachedDashboard(id: string) {
  return cache.dashboards.get(id)
}
```

### 4.4 自动更新

```typescript
// electron/services/updater.ts
import { autoUpdater } from 'electron-updater'
import { BrowserWindow, ipcMain } from 'electron'

export function initUpdater(mainWindow: BrowserWindow) {
  autoUpdater.autoDownload = false
  autoUpdater.autoInstallOnAppQuit = true

  autoUpdater.on('checking-for-update', () => {
    mainWindow.webContents.send('update:checking')
  })

  autoUpdater.on('update-available', (info) => {
    mainWindow.webContents.send('update:available', info)
  })

  autoUpdater.on('update-not-available', () => {
    mainWindow.webContents.send('update:notAvailable')
  })

  autoUpdater.on('download-progress', (progress) => {
    mainWindow.webContents.send('update:progress', progress)
  })

  autoUpdater.on('update-downloaded', () => {
    mainWindow.webContents.send('update:downloaded')
  })

  autoUpdater.on('error', (err) => {
    mainWindow.webContents.send('update:error', err.message)
  })

  // IPC 处理
  ipcMain.handle('update:check', () => autoUpdater.checkForUpdates())
  ipcMain.handle('update:download', () => autoUpdater.downloadUpdate())
  ipcMain.handle('update:install', () => autoUpdater.quitAndInstall())
}
```

### 4.5 多窗口/多屏

```typescript
// electron/ipc/window.ts
import { ipcMain, BrowserWindow, screen } from 'electron'

export function setupWindowIpc() {
  // 创建新窗口
  ipcMain.handle('window:create', (_, options) => {
    const win = new BrowserWindow({
      ...options,
      webPreferences: {
        preload: path.join(__dirname, '../preload.js'),
        contextIsolation: true
      }
    })
    
    if (options.url) {
      win.loadURL(options.url)
    }
    
    return win.id
  })
  
  // 获取所有窗口
  ipcMain.handle('window:getAll', () => {
    return BrowserWindow.getAllWindows().map(win => ({
      id: win.id,
      title: win.getTitle(),
      bounds: win.getBounds()
    }))
  })
  
  // 移动窗口到指定显示器
  ipcMain.handle('window:moveToDisplay', (_, windowId, displayIndex) => {
    const win = BrowserWindow.fromId(windowId)
    const displays = screen.getAllDisplays()
    
    if (win && displays[displayIndex]) {
      const display = displays[displayIndex]
      win.setBounds(display.bounds)
    }
  })
  
  // 全屏
  ipcMain.handle('window:setFullScreen', (_, windowId, fullscreen) => {
    const win = BrowserWindow.fromId(windowId)
    if (win) {
      win.setFullScreen(fullscreen)
    }
  })
}
```

### 4.6 本地文件拖拽

```typescript
// 渲染进程 - 拖拽导入
<template>
  <div
    class="drop-zone"
    @dragover.prevent
    @drop.prevent="onDrop"
  >
    拖拽文件到此处导入
  </div>
</template>

<script setup lang="ts">
async function onDrop(event: DragEvent) {
  const files = event.dataTransfer?.files
  if (!files) return
  
  for (const file of files) {
    // 通过 Electron API 获取文件路径
    const filePath = (file as any).path
    
    // 读取文件内容
    const content = await window.electronAPI.file.read(filePath)
    
    // 解析并导入
    if (file.name.endsWith('.csv')) {
      await importCSV(content)
    } else if (file.name.endsWith('.xlsx')) {
      await importExcel(filePath)
    } else if (file.name.endsWith('.json')) {
      await importJSON(content)
    }
  }
}
</script>
```

### 4.7 系统托盘

```typescript
// electron/services/tray.ts
import { Tray, Menu, nativeImage, BrowserWindow } from 'electron'
import path from 'path'

export function initTray(mainWindow: BrowserWindow): Tray {
  const icon = nativeImage.createFromPath(
    path.join(__dirname, '../resources/icon.png')
  ).resize({ width: 16, height: 16 })
  
  const tray = new Tray(icon)
  
  const contextMenu = Menu.buildFromTemplate([
    {
      label: '显示主窗口',
      click: () => {
        mainWindow.show()
        mainWindow.focus()
      }
    },
    {
      label: '新建窗口',
      submenu: [
        { label: '仪表板', click: () => createWindow('/dashboard') },
        { label: '大屏', click: () => createWindow('/screen') }
      ]
    },
    { type: 'separator' },
    {
      label: '检查更新',
      click: () => autoUpdater.checkForUpdates()
    },
    { type: 'separator' },
    {
      label: '退出',
      click: () => {
        mainWindow.destroy()
        app.quit()
      }
    }
  ])
  
  tray.setToolTip('数据可视化平台')
  tray.setContextMenu(contextMenu)
  
  tray.on('click', () => {
    if (mainWindow.isVisible()) {
      mainWindow.hide()
    } else {
      mainWindow.show()
      mainWindow.focus()
    }
  })
  
  return tray
}
```
