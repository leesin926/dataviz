## 一、概述

### 1.1 定位

大屏端是系统的核心展示终端，面向指挥中心、监控室、展厅等场景。提供拼接屏、多屏同步、4K/8K 适配、远程控制、定时轮播、实时刷新、离线包、应急插播等专业大屏能力。

### 1.2 核心能力

- 拼接屏显示
- 多屏同步
- 4K/8K 超高清适配
- 远程控制
- 定时轮播
- 实时数据刷新
- 离线运行包
- 应急插播
- 高性能渲染

### 4.1 分辨率适配

```typescript
// core/adapter/ResolutionAdapter.ts
export class ResolutionAdapter {
  private designWidth: number
  private designHeight: number
  private container: HTMLElement
  private mode: 'fit' | 'fill' | 'stretch' | 'fullWidth'

  constructor(options: {
    designWidth: number
    designHeight: number
    container: HTMLElement
    mode?: 'fit' | 'fill' | 'stretch' | 'fullWidth'
  }) {
    this.designWidth = options.designWidth
    this.designHeight = options.designHeight
    this.container = options.container
    this.mode = options.mode || 'fit'
  }

  // 计算缩放比例
  calculateScale(): { scaleX: number; scaleY: number; scale: number } {
    const containerWidth = this.container.clientWidth
    const containerHeight = this.container.clientHeight

    const scaleX = containerWidth / this.designWidth
    const scaleY = containerHeight / this.designHeight

    let scale: number
    switch (this.mode) {
      case 'fit':
        scale = Math.min(scaleX, scaleY)
        break
      case 'fill':
        scale = Math.max(scaleX, scaleY)
        break
      case 'stretch':
        scale = 1 // 分别使用 scaleX/Y
        break
      case 'fullWidth':
        scale = scaleX
        break
      default:
        scale = Math.min(scaleX, scaleY)
    }

    return { scaleX, scaleY, scale }
  }

  // 应用适配
  apply() {
    const { scaleX, scaleY, scale } = this.calculateScale()

    if (this.mode === 'stretch') {
      this.container.style.transform = `scale(${scaleX}, ${scaleY})`
    } else {
      this.container.style.transform = `scale(${scale})`
    }

    // 居中
    const containerWidth = this.container.clientWidth * scale
    const containerHeight = this.container.clientHeight * scale
    const parentWidth = this.container.parentElement?.clientWidth || 0
    const parentHeight = this.container.parentElement?.clientHeight || 0

    this.container.style.left = `${(parentWidth - containerWidth) / 2}px`
    this.container.style.top = `${(parentHeight - containerHeight) / 2}px`
  }

  // 监听窗口变化
  watchResize(callback?: () => void) {
    const observer = new ResizeObserver(() => {
      this.apply()
      callback?.()
    })

    if (this.container.parentElement) {
      observer.observe(this.container.parentElement)
    }

    return () => observer.disconnect()
  }
}

// 使用
const adapter = new ResolutionAdapter({
  designWidth: 1920,
  designHeight: 1080,
  container: document.getElementById('screen')!,
  mode: 'fit'
})

adapter.apply()
adapter.watchResize()
```

```typescript
// core/adapter/MultiScreenAdapter.ts
export class MultiScreenAdapter {
  private totalRows: number
  private totalCols: number
  private currentRow: number
  private currentCol: number
  private totalWidth: number
  private totalHeight: number

  constructor(options: {
    totalRows: number
    totalCols: number
    currentRow: number  // 从 1 开始
    currentCol: number  // 从 1 开始
    totalWidth: number
    totalHeight: number
  }) {
    this.totalRows = options.totalRows
    this.totalCols = options.totalCols
    this.currentRow = options.currentRow
    this.currentCol = options.currentCol
    this.totalWidth = options.totalWidth
    this.totalHeight = options.totalHeight
  }

  // 计算当前屏幕的可视区域
  getViewport(): {
    x: number
    y: number
    width: number
    height: number
  } {
    const singleWidth = this.totalWidth / this.totalCols
    const singleHeight = this.totalHeight / this.totalRows

    return {
      x: (this.currentCol - 1) * singleWidth,
      y: (this.currentRow - 1) * singleHeight,
      width: singleWidth,
      height: singleHeight
    }
  }

  // 应用拼接裁剪
  apply(container: HTMLElement) {
    const viewport = this.getViewport()

    // 设置容器大小为整个大屏尺寸
    container.style.width = `${this.totalWidth}px`
    container.style.height = `${this.totalHeight}px`

    // 通过 transform 偏移到当前屏幕的位置
    container.style.transform = `translate(-${viewport.x}px, -${viewport.y}px)`

    // 父容器裁剪
    const parent = container.parentElement
    if (parent) {
      parent.style.width = `${viewport.width}px`
      parent.style.height = `${viewport.height}px`
      parent.style.overflow = 'hidden'
    }
  }
}

// 使用 - 3x3 拼接屏中的第 2 行第 2 列 (中间屏幕)
const adapter = new MultiScreenAdapter({
  totalRows: 3,
  totalCols: 3,
  currentRow: 2,
  currentCol: 2,
  totalWidth: 5760,  // 3 * 1920
  totalHeight: 3240  // 3 * 1080
})

adapter.apply(document.getElementById('screen')!)
```

### 4.3 实时数据引擎

```typescript
// core/data/DataFetcher.ts
export class DataFetcher {
  private fetchers = new Map<string, DataFetcherInstance>()
  private cache = new DataCache()

  // 注册数据源
  register(componentId: string, config: DataSourceConfig) {
    const fetcher = this.createFetcher(config)
    this.fetchers.set(componentId, fetcher)
    return fetcher
  }

  // 创建数据获取器
  private createFetcher(config: DataSourceConfig): DataFetcherInstance {
    switch (config.type) {
      case 'api':
        return new ApiFetcher(config)
      case 'websocket':
        return new WebSocketFetcher(config)
      case 'database':
        return new DatabaseFetcher(config)
      case 'polling':
        return new PollingFetcher(config)
      default:
        return new StaticFetcher(config)
    }
  }

  // 获取数据
  async fetch(componentId: string): Promise<any> {
    const fetcher = this.fetchers.get(componentId)
    if (!fetcher) throw new Error(`Fetcher not found: ${componentId}`)

    // 检查缓存
    const cacheKey = this.getCacheKey(componentId, fetcher.config)
    const cached = this.cache.get(cacheKey)
    if (cached && !this.isExpired(cached, fetcher.config.ttl)) {
      return cached.data
    }

    // 获取新数据
    const data = await fetcher.fetch()

    // 更新缓存
    this.cache.set(cacheKey, { data, timestamp: Date.now() })

    return data
  }

  // 启动实时数据
  startRealtime(componentId: string, callback: (data: any) => void) {
    const fetcher = this.fetchers.get(componentId)
    if (!fetcher) return

    if (fetcher instanceof WebSocketFetcher) {
      fetcher.onMessage(callback)
      fetcher.connect()
    } else if (fetcher instanceof PollingFetcher) {
      fetcher.startPolling(callback)
    }
  }

  // 停止实时数据
  stopRealtime(componentId: string) {
    const fetcher = this.fetchers.get(componentId)
    if (!fetcher) return

    if (fetcher instanceof WebSocketFetcher) {
      fetcher.disconnect()
    } else if (fetcher instanceof PollingFetcher) {
      fetcher.stopPolling()
    }
  }
}

// WebSocket 数据获取器
class WebSocketFetcher implements DataFetcherInstance {
  private ws: WebSocket | null = null
  private messageCallback: ((data: any) => void) | null = null

  config: DataSourceConfig

  constructor(config: DataSourceConfig) {
    this.config = config
  }

  async fetch(): Promise<any> {
    return new Promise((resolve) => {
      this.onMessage(resolve)
      this.connect()
    })
  }

  connect() {
    if (this.ws?.readyState === WebSocket.OPEN) return

    this.ws = new WebSocket(this.config.wsUrl!)

    this.ws.onopen = () => {
      // 发送订阅消息
      if (this.config.wsMessage) {
        this.ws?.send(this.config.wsMessage)
      }
    }

    this.ws.onmessage = (event) => {
      const data = JSON.parse(event.data)
      this.messageCallback?.(data)
    }

    this.ws.onerror = (error) => {
      console.error('WebSocket error:', error)
    }

    this.ws.onclose = () => {
      // 自动重连
      setTimeout(() => this.connect(), 3000)
    }
  }

  disconnect() {
    this.ws?.close()
    this.ws = null
  }

  onMessage(callback: (data: any) => void) {
    this.messageCallback = callback
  }
}

// 轮询数据获取器
class PollingFetcher implements DataFetcherInstance {
  private timer: ReturnType<typeof setInterval> | null = null
  private callback: ((data: any) => void) | null = null

  config: DataSourceConfig

  constructor(config: DataSourceConfig) {
    this.config = config
  }

  async fetch(): Promise<any> {
    const response = await fetch(this.config.apiUrl!, {
      headers: this.config.apiHeaders
    })
    return response.json()
  }

  startPolling(callback: (data: any) => void) {
    this.callback = callback
    const interval = this.config.refreshInterval || 5000

    this.timer = setInterval(async () => {
      const data = await this.fetch()
      callback(data)
    }, interval)
  }

  stopPolling() {
    if (this.timer) {
      clearInterval(this.timer)
      this.timer = null
    }
  }
}
```

### 4.4 轮播调度

```typescript
// core/scheduler/CarouselScheduler.ts
export class CarouselScheduler {
  private pages: ScreenPage[] = []
  private currentIndex = 0
  private timer: ReturnType<typeof setInterval> | null = null
  private isPlaying = false

  constructor(private onPageChange: (index: number) => void) {}

  // 设置页面列表
  setPages(pages: ScreenPage[]) {
    this.pages = pages
  }

  // 开始轮播
  start() {
    if (this.isPlaying || this.pages.length <= 1) return

    this.isPlaying = true
    this.scheduleNext()
  }

  // 停止轮播
  stop() {
    this.isPlaying = false
    if (this.timer) {
      clearTimeout(this.timer)
      this.timer = null
    }
  }

  // 调度下一页
  private scheduleNext() {
    if (!this.isPlaying) return

    const currentPage = this.pages[this.currentIndex]
    const duration = currentPage.duration || 10000

    this.timer = setTimeout(() => {
      this.next()
      this.scheduleNext()
    }, duration)
  }

  // 下一页
  next() {
    this.currentIndex = (this.currentIndex + 1) % this.pages.length
    this.onPageChange(this.currentIndex)
  }

  // 上一页
  prev() {
    this.currentIndex = (this.currentIndex - 1 + this.pages.length) % this.pages.length
    this.onPageChange(this.currentIndex)
  }

  // 跳转到指定页
  goTo(index: number) {
    if (index >= 0 && index < this.pages.length) {
      this.currentIndex = index
      this.onPageChange(this.currentIndex)

      // 重新调度
      if (this.isPlaying) {
        this.stop()
        this.start()
      }
    }
  }

  // 获取当前页索引
  getCurrentIndex(): number {
    return this.currentIndex
  }

  // 是否正在播放
  getIsPlaying(): boolean {
    return this.isPlaying
  }
}
```

### 4.5 远程控制

```typescript
// composables/useScreen.ts
export function useScreen(screenId: string) {
  const ws = ref<WebSocket>()
  const isConnected = ref(false)
  const currentSlide = ref(0)
  const isAutoplay = ref(false)

  // 连接控制服务
  function connect() {
    ws.value = new WebSocket(`${WS_URL}/screen/control/${screenId}`)

    ws.value.onopen = () => {
      isConnected.value = true
      console.log('Screen control connected')
    }

    ws.value.onmessage = (event) => {
      const command = JSON.parse(event.data)
      handleCommand(command)
    }

    ws.value.onclose = () => {
      isConnected.value = false
      // 自动重连
      setTimeout(() => connect(), 3000)
    }
  }

  // 处理控制指令
  function handleCommand(command: ScreenCommand) {
    switch (command.action) {
      case 'switch':
        currentSlide.value = command.slide
        break
      case 'next':
        currentSlide.value++
        break
      case 'prev':
        currentSlide.value--
        break
      case 'autoplay':
        isAutoplay.value = command.enabled
        break
      case 'refresh':
        location.reload()
        break
      case 'emergency':
        showEmergency(command.content)
        break
      case 'volume':
        setVolume(command.level)
        break
      case 'fullscreen':
        toggleFullscreen()
        break
    }
  }

  // 应急插播
  function showEmergency(content: string) {
    // 显示紧急内容
    const overlay = document.createElement('div')
    overlay.className = 'emergency-overlay'
    overlay.innerHTML = content
    document.body.appendChild(overlay)

    // 10秒后自动关闭
    setTimeout(() => {
      overlay.remove()
    }, 10000)
  }

  // 上报状态
  function reportStatus(status: ScreenStatus) {
    ws.value?.send(JSON.stringify({
      type: 'status',
      screenId,
      ...status
    }))
  }

  return {
    connect,
    isConnected,
    currentSlide,
    isAutoplay,
    reportStatus
  }
}
```

```typescript
// composables/usePerformance.ts
export function usePerformance() {
  const fps = ref(0)
  const memory = ref(0)
  const renderTime = ref(0)

  // 监控 FPS
  function monitorFPS() {
    let frameCount = 0
    let lastTime = performance.now()

    function loop() {
      frameCount++
      const now = performance.now()

      if (now - lastTime >= 1000) {
        fps.value = frameCount
        frameCount = 0
        lastTime = now

        // FPS 过低时降级
        if (fps.value < 30) {
          enableDegradation()
        }
      }

      requestAnimationFrame(loop)
    }

    requestAnimationFrame(loop)
  }

  // 启用降级策略
  function enableDegradation() {
    console.warn('Performance degradation detected, enabling fallback')

    // 降低刷新频率
    // 禁用粒子效果
    // 简化动画
    // 减少同时渲染的组件数
  }

  // 组件懒加载
  function lazyLoadComponent(componentId: string) {
    const observer = new IntersectionObserver(
      (entries) => {
        entries.forEach((entry) => {
          if (entry.isIntersecting) {
            loadComponent(componentId)
            observer.unobserve(entry.target)
          }
        })
      },
      { threshold: 0.1 }
    )

    const element = document.getElementById(componentId)
    if (element) {
      observer.observe(element)
    }
  }

  // GPU 加速
  function enableGPUAcceleration(element: HTMLElement) {
    element.style.transform = 'translateZ(0)'
    element.style.willChange = 'transform'
  }

  // 数据增量更新
  function incrementalUpdate(
    oldData: any[],
    newData: any[],
    keyField: string
  ) {
    const changes: DataChange[] = []

    const oldMap = new Map(oldData.map(d => [d[keyField], d]))
    const newMap = new Map(newData.map(d => [d[keyField], d]))

    // 新增
    for (const [key, value] of newMap) {
      if (!oldMap.has(key)) {
        changes.push({ type: 'add', data: value })
      }
    }

    // 更新
    for (const [key, value] of newMap) {
      if (oldMap.has(key) && !isEqual(oldMap.get(key), value)) {
        changes.push({ type: 'update', key, data: value })
      }
    }

    // 删除
    for (const [key] of oldMap) {
      if (!newMap.has(key)) {
        changes.push({ type: 'delete', key })
      }
    }

    return changes
  }

  return {
    fps,
    memory,
    renderTime,
    monitorFPS,
    enableDegradation,
    lazyLoadComponent,
    enableGPUAcceleration,
    incrementalUpdate
  }
}
```

```typescript
// 生成离线包
export async function generateOfflinePackage(screenId: string) {
  // 1. 获取大屏配置
  const screen = await screenApi.get(screenId)

  // 2. 预加载所有资源
  const resources = await preloadResources(screen)

  // 3. 预加载数据
  const data = await preloadData(screen)

  // 4. 打包
  const package = {
    version: screen.version,
    generatedAt: Date.now(),
    screen,
    resources,
    data
  }

  // 5. 压缩
  const compressed = await compress(package)

  // 6. 下载
  downloadBlob(compressed, `screen-${screenId}-offline.zip`)
}

// 加载离线包
export async function loadOfflinePackage(file: File) {
  const zip = new JSZip()
  const content = await zip.loadAsync(file)

  const packageJson = await content.file('package.json')?.async('text')
  const package = JSON.parse(packageJson!)

  // 恢复大屏配置
  const screenStore = useScreenStore()
  screenStore.setScreen(package.screen)

  // 恢复数据
  const dataStore = useDataStore()
  dataStore.setCachedData(package.data)

  // 恢复资源
  await restoreResources(package.resources)
}
```
