## 一、概述

### 1.1 定位

小程序端面向移动轻量场景，支持微信、钉钉、飞书、支付宝多平台。提供扫码看数、轻量看板、审批待办、移动大屏控制等功能，是用户随时随地查看数据的核心入口。

### 1.2 核心能力

- 多端适配（微信/钉钉/飞书/支付宝）
- 扫码看数
- 分享卡片
- 订阅消息
- 轻量看板
- 审批待办
- 移动大屏控制

```typescript
// composables/useAuth.ts
export function useAuth() {
  const userStore = useUserStore()
  
  async function login() {
    // 1. 获取微信 code
    const { code } = await uni.login({ provider: 'weixin' })
    
    // 2. 调用后端换取 token
    const { token, userInfo } = await authApi.wechatLogin({ code })
    
    // 3. 存储用户信息
    userStore.setToken(token)
    userStore.setUserInfo(userInfo)
    
    return userInfo
  }
  
  async function logout() {
    await authApi.logout()
    userStore.clear()
    uni.reLaunch({ url: '/pages/login/index' })
  }
  
  return { login, logout }
}

// api/modules/auth.ts
export const authApi = {
  wechatLogin: (data: { code: string }) =>
    request.post<LoginResult>('/auth/wechat', data),
  
  dingtalkLogin: (data: { authCode: string }) =>
    request.post<LoginResult>('/auth/dingtalk', data),
  
  feishuLogin: (data: { code: string }) =>
    request.post<LoginResult>('/auth/feishu', data),
  
  alipayLogin: (data: { authCode: string }) =>
    request.post<LoginResult>('/auth/alipay', data)
}
```

### 5.2 扫码看数

```typescript
// pages/scan/index.vue
<template>
  <view class="scan-page">
    <scan-line @success="onScanSuccess" />
  </view>
</template>

<script setup lang="ts">
function onScanSuccess(result: string) {
  // 解析扫码结果
  // 格式: dataviz://dashboard/{id}
  //       dataviz://screen/{id}
  
  const parsed = parseQRCode(result)
  
  if (parsed.type === 'dashboard') {
    uni.navigateTo({
      url: `/pages/dashboard-view/index?id=${parsed.id}`
    })
  } else if (parsed.type === 'screen') {
    uni.navigateTo({
      url: `/pages/screen-view/index?id=${parsed.id}`
    })
  }
}

function parseQRCode(code: string) {
  // 支持多种格式
  if (code.startsWith('dataviz://')) {
    const [, type, id] = code.split('/')
    return { type, id }
  }
  
  // URL 格式
  const url = new URL(code)
  if (url.pathname.includes('/dashboard/')) {
    return { type: 'dashboard', id: url.pathname.split('/').pop() }
  }
  if (url.pathname.includes('/screen/')) {
    return { type: 'screen', id: url.pathname.split('/').pop() }
  }
  
  throw new Error('无效的二维码')
}
</script>
```

```vue
    uni.stopPullDownRefresh()
  })
})
```

### 5.4 图表组件

```vue
<!-- components/chart/ChartRenderer.vue -->
<template>
  <view class="chart-renderer">
    <view class="chart-title" v-if="config.title">
      {{ config.title }}
    </view>
    
    <!-- KPI 卡片 -->
    <KpiCard
      v-if="type === 'kpi'"
      :data="data"
      :config="config"
    />
    
    <!-- 图表 -->
    <view v-else class="chart-container">
      <qiun-data-charts
        :type="chartType"
        :chartData="chartData"
        :opts="chartOpts"
        :ontouch="true"
        @tap="onChartTap"
      />
    </view>
  </view>
</template>

<script setup lang="ts">
import { computed } from 'vue'

interface Props {
  type: string
  config: ChartConfig
  data: any
}

const props = defineProps<Props>()

// 映射到小程序图表类型
const chartType = computed(() => {
  const typeMap: Record<string, string> = {
    'line': 'line',
    'bar': 'column',
    'pie': 'pie',
    'area': 'area',
    'radar': 'radar',
    'funnel': 'funnel'
  }
  return typeMap[props.type] || 'column'
})

const chartData = computed(() => {
  // 转换为 uCharts 格式
  return transformData(props.data, props.type)
})

const chartOpts = computed(() => {
  return {
    color: props.config.colors || ['#1890ff', '#2fc25b', '#facc14'],
    title: {
      name: props.config.title || ''
    },
    legend: {
      show: props.config.showLegend !== false
    },
    extra: {
      [chartType.value]: {
        // 类型特定配置
      }
    }
  }
})

function transformData(data: any, type: string) {
  // 根据图表类型转换数据格式
  if (type === 'kpi') {
    return data
  }
  
  // 通用转换
  return {
    categories: data.map((d: any) => d.category),
    series: [{
      name: data.name || '数据',
      data: data.map((d: any) => d.value)
    }]
  }
}
</script>
```

```typescript
    // 绘制图表截图
```

```typescript
// composables/useSubscribe.ts
export function useSubscribe() {
  // 请求订阅
  async function requestSubscribe(templateIds: string[]) {
    // #ifdef MP-WEIXIN
    const result = await uni.requestSubscribeMessage({
      tmplIds: templateIds
    })
    
    if (result.errMsg.includes('accept')) {
      uni.showToast({ title: '订阅成功', icon: 'success' })
      return true
    }
    // #endif
    
    return false
  }
  
  // 订阅告警通知
  async function subscribeAlert(dashboardId: string) {
    const templateId = 'TEMPLATE_ALERT_ID'
    const accepted = await requestSubscribe([templateId])
    
    if (accepted) {
      await alertApi.subscribe({
        dashboardId,
        channel: 'miniprogram',
        templateId
      })
    }
  }
  
  // 订阅审批通知
  async function subscribeApproval() {
    const templateId = 'TEMPLATE_APPROVAL_ID'
    await requestSubscribe([templateId])
  }
  
  return { requestSubscribe, subscribeAlert, subscribeApproval }
}
```

```vue
const todoList = ref<TodoItem[]>([])
const page = ref(1)
const loadingStatus = ref<'more' | 'loading' | 'noMore'>('more')

async function loadTodos() {
  loadingStatus.value = 'loading'
  
  const types = ['pending', 'processed', 'initiated']
  const result = await todoApi.list({
    type: types[activeTab.value],
    page: page.value,
    size: 20
  })
  
  if (page.value === 1) {
    todoList.value = result.records
  } else {
    todoList.value.push(...result.records)
  }
  
  loadingStatus.value = result.records.length < 20 ? 'noMore' : 'more'
}

function loadMore() {
  if (loadingStatus.value === 'more') {
    page.value++
    loadTodos()
  }
}

function goToDetail(item: TodoItem) {
  uni.navigateTo({
    url: `/pages/todo/detail?id=${item.id}&type=${item.type}`
  })
}
</script>
```

```typescript
// composables/useScreenControl.ts
export function useScreenControl(screenId: string) {
  const ws = ref<UniApp.SocketTask>()
  const currentSlide = ref(0)
  const isConnected = ref(false)
  
  // 连接 WebSocket
  function connect() {
    ws.value = uni.connectSocket({
      url: `${WS_URL}/screen/control/${screenId}`,
      success: () => {
        isConnected.value = true
      }
    })
    
    ws.value?.onMessage((res) => {
      const data = JSON.parse(res.data)
      handleScreenMessage(data)
    })
  }
  
  // 发送控制指令
  function sendCommand(command: ScreenCommand) {
    ws.value?.send({
      data: JSON.stringify({
        type: 'control',
        command,
        timestamp: Date.now()
      })
    })
  }
  
  // 切换页面
  function switchSlide(index: number) {
    sendCommand({ action: 'switch', slide: index })
    currentSlide.value = index
  }
  
  // 上一页
  function prevSlide() {
    if (currentSlide.value > 0) {
      switchSlide(currentSlide.value - 1)
    }
  }
  
  // 下一页
  function nextSlide() {
    switchSlide(currentSlide.value + 1)
  }
  
  // 暂停/恢复轮播
  function toggleAutoplay() {
    sendCommand({ action: 'toggleAutoplay' })
  }
  
  // 紧急插播
  function emergencyInsert(content: string) {
    sendCommand({ action: 'emergency', content })
  }
  
  // 断开连接
  function disconnect() {
    ws.value?.close({})
    isConnected.value = false
  }
  
  return {
    connect,
    disconnect,
    switchSlide,
    prevSlide,
    nextSlide,
    toggleAutoplay,
    emergencyInsert,
    currentSlide,
    isConnected
  }
}
```
